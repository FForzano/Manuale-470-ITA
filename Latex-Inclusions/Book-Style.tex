\pagestyle{fancy}
\fancyhf{} % cancella header/footer predefiniti

% Contenuto: capitolo a sinistra sulle pagine pari, sezione a destra sulle pagine dispari
\fancyhead[LE]{\small\bfseries\leftmark} % Left header on Even pages: chapter
\fancyhead[RO]{\small\bfseries\rightmark} % Right header on Odd pages: section

% Numero di pagina "all'esterno" (outside)
\fancyfoot[LE,RO]{\small\thepage}

% linea sotto l'header e niente linea sopra il footer
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}

% Controllo di cosa contengono \leftmark e \rightmark
\renewcommand{\chaptermark}[1]{\markboth{\thechapter\ \ #1}{}}
\renewcommand{\sectionmark}[1]{\markright{\thesection\ \ #1}}

% Personalizza lo stile "plain" (usato dalle pagine di inizio capitolo)
\fancypagestyle{plain}{
  \fancyhf{} % cancella
  \fancyfoot[LE,RO]{\small\thepage} % mantiene il numero di pagina
  \renewcommand{\headrulewidth}{0pt} % no linea in pagine plain
}
